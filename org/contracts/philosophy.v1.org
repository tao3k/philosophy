        (= (property "PRINCIPLE_KIND") "foundational")
        (= (property "PRINCIPLE_KIND") "refinement"))
      (not (= (property "NORMATIVE_LEVEL") ""))
      (not (= (property "SCOPE") ""))
      (not (= (property "REVISION") ""))
      (not (= (property "REFINES") ""))
